# Programa de conversão de moedas

# No campo abaixo o usuário deve digitar a moeda que deseja converter (dolar, euro ou iene). 
ls_moeda = input("Digite a moeda que deseja converter: (dolar, euro ou iene): ").lower() #O método lower() é utilizado para converter a entrada do usuário para minúsculas.

# Com a estrutura abaixo o programa irá solicitar um valor em reais e realizar a conversão para a moeda escolhida.

if ls_moeda == "dolar": #Se moeda for igual a "dolar", o programa irá solicitar um valor em reais e realizar a conversão para dólares.
    lf_valor = float(input("Digite o valor em reais: "))
    valor_convertido = lf_valor / 5.25 #A taxa de conversão utilizada é de 1 dólar = 5.25 reais. 
    print(f"O valor em dólares é: {valor_convertido:.2f}") #O "f" antes das aspas permite a formatação de variáveis dentro da string, já o valor_convertido:.2f formata o valor para duas casas decimais.

elif ls_moeda == "euro": #Se moeda for igual a "euro", o programa irá solicitar um valor em reais e realizar a conversão para euros.
    lf_valor = float(input("Digite o valor em reais: "))
    valor_convertido = lf_valor / 6.20 #A taxa de conversão utilizada é de 1 euro = 6.20 reais. 
    print(f"O valor em euros é: {valor_convertido:.2f}")

elif ls_moeda == "iene": #Se moeda for igual a "iene", o programa irá solicitar um valor em reais e realizar a conversão para ienes.
    lf_valor = float(input("Digite o valor em reais: "))
    valor_convertido = lf_valor / 0.048 #A taxa de conversão utilizada é de 1 iene = 0.048 reais.
    print(f"O valor em ienes é: {valor_convertido:.2f}")

else: #Se o usuário digitar uma moeda diferente de "dolar", "euro" ou "iene", o programa irá exibir uma mensagem de erro.
    print("Moeda inválida. Por favor, digite 'dolar', 'euro' ou 'iene'.")

# Controle de estoque

# Criação de um dicionário para armazenar os produtos e suas quantidades.
ld_produtos = {
} # o dicionário é inicialmente vazio, mas os produtos e suas quantidades serão adicionados conforme o usuário interagir com o programa. 


while True: # laço de repetição para permitir que o usuário realize várias operações até decidir sair do programa.
    
    # exibe o estoque atual, mostrando cada produto e sua quantidade.
    print("\nEstoque atual:") # o "\n" é utilizado para criar uma nova linha.
        for produto, quantidade in ld_produtos.items(): # o método items() é utilizado para obter os itens do dicionário correspondente.
        print(f"{produto}: {quantidade}") # aqui é exibido o nome do produto e a quantidade correspondente.
    print()

    # aqui foi utilizado uma série de prints para mostrar as opções disponíveis para o usuário com clareza, e deixar o programa mais organizado e fácil de usar.
    print("Digite a operação que deseja realizar:") # o usuário é orientado a escolher uma ação pré-definida, onde cada uma delas resulta em uma operação diferente.
    print("1 - Adicionar produto") # o usuario pode adicionar um produto ao estoque.
    print("2 - Remover produto") # o usuario pode remover um produto do estoque.
    print("3 - Verificar estoque") # o usuario pode verificar o estoque atual.
    print("4 - Sair") # o usuario encerra o programa.
    li_opcao = int(input("Opção: ")) 

    if li_opcao == 1: # caso o usuário escolha a opção 1 a seguinte operação será realizada:
        ls_produto = input("\nDigite o nome do produto que deseja adicionar: ").lower() # o usuario informa qual produto deseja adicionar ao estoque
        li_quantidade = int(input("Digite a quantidade do produto: ")) # logo em seguida informa a quantidade desse produto que vai ir para o estoque
        if ls_produto in ld_produtos: # o programa verifica se o produto já existe no estoque.
            ld_produtos[ls_produto] += li_quantidade # se o produto já existir, a quantidade informada é adicionada à quantidade existente no estoque
        else:
            ld_produtos[ls_produto] = li_quantidade # se o produto não existir, ele é adicionado ao dicionário com a quantidade informada
        print(f"Produto '{ls_produto}' adicionado com sucesso! Quantidade atual: {ld_produtos[ls_produto]}") # o programa confirma que o produto foi adicionado com sucesso e exibe a quantidade atual do produto

    elif li_opcao == 2: # caso o usuário escolha a opção 2 a seguinte operação será realizada:
        ls_produto = input("\nDigite o nome do produto que deseja remover: ").lower() # o usuario informa qual produto deseja remover do estoque
        if ls_produto in ld_produtos: # o programa verifica se o produto existe no estoque
            li_quantidade = int(input("Digite a quantidade do produto a ser removida: ")) # o usuario informa a quantidade do produto que deseja remover do estoque
            if li_quantidade <= ld_produtos[ls_produto]: # o programa verifica se a quantidade a ser removida é menor ou igual à quantidade disponível no estoque
                ld_produtos[ls_produto] -= li_quantidade # se a quantidade a ser removida for válida, ela é subtraída da quantidade atual do produto no estoque
                print(f"Produto '{ls_produto}' removido com sucesso! Quantidade atual: {ld_produtos[ls_produto]}") #o codigo ira imprimir a a confirmacao de que o produto foi removido com sucesso e exibe a quantidade atual do produto no estoque
            else:
                print(f"Quantidade insuficiente em estoque. Quantidade atual de '{ls_produto}': {ld_produtos[ls_produto]}") # se a quantidade a ser removida for maior do que a quantidade disponível no estoque, o programa exibe uma mensagem de erro.
        else:
            print(f"Produto '{ls_produto}' não encontrado no estoque.") # se o produto informado para remoção não existir no estoque, o programa exibe uma mensagem de erro.

    elif li_opcao == 3: # caso o usuário escolha a opção 3 a seguinte operação será realizada:
        print("\nEstoque atual:") # o programa exibe o estoque atual, mostrando cada produto e sua quantidade.
        for produto, quantidade in ld_produtos.items(): # o método items() é utilizado para obter os itens do dicionário correspondente, e exibe o nome do produto e a quantidade correspondente.
            print(f"\n{produto}: {quantidade}") # exibe o nome do produto e a quantidade correspondente.

        if not ld_produtos: # se o dicionário de produtos estiver vazio o programa exibe uma mensagem de alerta
            print("O estoque está vazio.")

    elif li_opcao == 4: # caso o usuário escolha a opção 4 a seguinte operação será realizada:
        print("\nEncerrando o programa. Até mais!") # o programa exibe uma mensagem de despedida e encerra o programa utilizando o comando break.
        break # encerra o loop, e consequentemente o programa.

}

ld_lucro = {}

# aqui foi usado um laço de repetição para que o usúario possa realizar várias operações sem precisar reiniciar o programa.
while True:
    print("\nSistema de Controle de Vendas, qual operação deseja realizar?") # aqui o programa pergunta ao usuário qual operação ele deseja realizar
    print("1 - Registrar venda") # o usuario pode registrar uma venda, onde ele irá informar o nome do produto vendido e o preço de venda, e o programa irá calcular o lucro obtido com a venda
    print("2 - Verificar estoque") # o usuario pode verificar o estoque, onde o programa irá exibir os produtos disponíveis para venda e seus preços
    print("3 - Sair") # e o usuario pode encerrar o loop, e consequentemente o programa
    li_opcao = input("Opção: ") # aqui fica o campo onde o usuário registra sua decisão

    if li_opcao == "1": # caso o usuário escolha a opção 1 a seguinte operação será realizada:
        ls_produto = input("Digite o nome do produto vendido: ") # o usuario informa o nome do produto vendido

            print(f"Lucro obtido: R${lf_lucro:.2f}") # o programa exibe o lucro obtido com a venda, formatado para duas casas decimais utilizando :.2f
        else: # se o produto informado pelo usuário não existe no dicionário de produtos
            print("Produto não encontrado.") # o programa exibe uma mensagem de erro

        print("Encerrando o programa. Até mais!") # o programa exibe uma mensagem
        break # e logo em seguida encerra o programa utilizando o comando break, que encerra o loop do while

    else: # se o usuário digitar uma opção inválida, ou seja, diferente de 1, 2 ou 3
        print("Opção inválida. Por favor, escolha uma opção válida.") # o programa exibe uma mensagem de erro, orientando o usuário a escolher uma opção válida

# Criação de um dicionário para armazenar os produtos e suas quantidades.
ld_produtos = {
} # o dicionário é inicialmente vazio, mas os produtos e suas quantidades serão adicionados conforme o usuário interagir com o programa. 


while True: # laço de repetição para permitir que o usuário realize várias operações até decidir sair do programa.
    
    # exibe o estoque atual, mostrando cada produto e sua quantidade.
    print("\nEstoque atual:") # o "\n" é utilizado para criar uma nova linha.
    for produto, quantidade in ld_produtos.items(): # o método items() é utilizado para obter os itens do dicionário correspondente.
        print(f"{produto}: {quantidade}") # aqui é exibido o nome do produto e a quantidade correspondente.
    print()

    # aqui foi utilizado uma série de prints para mostrar as opções disponíveis para o usuário com clareza, e deixar o programa mais organizado e fácil de usar.
    print("Digite a operação que deseja realizar:") # o usuário é orientado a escolher uma ação pré-definida, onde cada uma delas resulta em uma operação diferente.
    print("1 - Adicionar produto") # o usuario pode adicionar um produto ao estoque.
    print("2 - Remover produto") # o usuario pode remover um produto do estoque.
    print("3 - Verificar estoque") # o usuario pode verificar o estoque atual.
    print("4 - Sair") # o usuario encerra o programa.
    li_opcao = int(input("Opção: ")) 

    if li_opcao == 1: # caso o usuário escolha a opção 1 a seguinte operação será realizada:
        ls_produto = input("\nDigite o nome do produto que deseja adicionar: ").lower() # o usuario informa qual produto deseja adicionar ao estoque
        li_quantidade = int(input("Digite a quantidade do produto: ")) # logo em seguida informa a quantidade desse produto que vai ir para o estoque
        if ls_produto in ld_produtos: # o programa verifica se o produto já existe no estoque.
            ld_produtos[ls_produto] += li_quantidade # se o produto já existir, a quantidade informada é adicionada à quantidade existente no estoque
        else:
            ld_produtos[ls_produto] = li_quantidade # se o produto não existir, ele é adicionado ao dicionário com a quantidade informada
        print(f"Produto '{ls_produto}' adicionado com sucesso! Quantidade atual: {ld_produtos[ls_produto]}") # o programa confirma que o produto foi adicionado com sucesso e exibe a quantidade atual do produto

    elif li_opcao == 2: # caso o usuário escolha a opção 2 a seguinte operação será realizada:
        ls_produto = input("\nDigite o nome do produto que deseja remover: ").lower() # o usuario informa qual produto deseja remover do estoque
        if ls_produto in ld_produtos: # o programa verifica se o produto existe no estoque
            li_quantidade = int(input("Digite a quantidade do produto a ser removida: ")) # o usuario informa a quantidade do produto que deseja remover do estoque
            if li_quantidade <= ld_produtos[ls_produto]: # o programa verifica se a quantidade a ser removida é menor ou igual à quantidade disponível no estoque
                ld_produtos[ls_produto] -= li_quantidade # se a quantidade a ser removida for válida, ela é subtraída da quantidade atual do produto no estoque
                print(f"Produto '{ls_produto}' removido com sucesso! Quantidade atual: {ld_produtos[ls_produto]}") #o codigo ira imprimir a a confirmacao de que o produto foi removido com sucesso e exibe a quantidade atual do produto no estoque
            else:
                print(f"Quantidade insuficiente em estoque. Quantidade atual de '{ls_produto}': {ld_produtos[ls_produto]}") # se a quantidade a ser removida for maior do que a quantidade disponível no estoque, o programa exibe uma mensagem de erro.
        else:
            print(f"Produto '{ls_produto}' não encontrado no estoque.") # se o produto informado para remoção não existir no estoque, o programa exibe uma mensagem de erro.

    elif li_opcao == 3: # caso o usuário escolha a opção 3 a seguinte operação será realizada:
        print("\nEstoque atual:") # o programa exibe o estoque atual, mostrando cada produto e sua quantidade.
        for produto, quantidade in ld_produtos.items(): # o método items() é utilizado para obter os itens do dicionário correspondente, e exibe o nome do produto e a quantidade correspondente.
            print(f"\n{produto}: {quantidade}") # exibe o nome do produto e a quantidade correspondente.

        if not ld_produtos: # se o dicionário de produtos estiver vazio o programa exibe uma mensagem de alerta
            print("O estoque está vazio.")

    elif li_opcao == 4: # caso o usuário escolha a opção 4 a seguinte operação será realizada:
        print("\nEncerrando o programa. Até mais!") # o programa exibe uma mensagem de despedida e encerra o programa utilizando o comando break.
        break # encerra o loop, e consequentemente o programa.

# Sistema de Controle de Vendas

# O campo abaixo cria um dicionário para armazenar os produtos e seus preços.
ld_produtos = {
    "Notebook": 1500.00, 
    "Smartphone": 800.00,
    "Tablet": 600.00
}

ld_lucro = {}

# aqui foi usado um laço de repetição para que o usúario possa realizar várias operações sem precisar reiniciar o programa.
while True:
    print("\nSistema de Controle de Vendas, qual operação deseja realizar?") # aqui o programa pergunta ao usuário qual operação ele deseja realizar
    print("1 - Registrar venda") # o usuario pode registrar uma venda, onde ele irá informar o nome do produto vendido e o preço de venda, e o programa irá calcular o lucro obtido com a venda
    print("2 - Verificar estoque") # o usuario pode verificar o estoque, onde o programa irá exibir os produtos disponíveis para venda e seus preços
    print("3 - Sair") # e o usuario pode encerrar o loop, e consequentemente o programa
    li_opcao = input("Opção: ") # aqui fica o campo onde o usuário registra sua decisão

    if li_opcao == "1": # caso o usuário escolha a opção 1 a seguinte operação será realizada:
        ls_produto = input("Digite o nome do produto vendido: ") # o usuario informa o nome do produto vendido
        lf_preco_venda = float(input("Digite o preço de venda: ")) # o usuario informa o preço de venda do produto
        lf_lucro = lf_preco_venda - ld_produtos[ls_produto] # o programa calcula o lucro obtido com a venda subtraindo o preço de venda do preço do produto registrado no dicionário
        ld_lucro[ls_produto] = lf_lucro # o programa armazena o lucro obtido no dicionário de lucros

        if ls_produto in ld_produtos: # se o produto informado pelo usuário existe no dicionário de produtos
            print(f"Venda registrada: {ls_produto} de R${ld_produtos[ls_produto]} por R${lf_preco_venda:.2f}") # o programa exibe uma mensagem confirmando que a venda foi registrada
            print(f"Lucro obtido: R${lf_lucro:.2f}") # o programa exibe o lucro obtido com a venda, formatado para duas casas decimais utilizando :.2f
        else: # se o produto informado pelo usuário não existe no dicionário de produtos
            print("Produto não encontrado.") # o programa exibe uma mensagem de erro

    elif li_opcao == "2": # caso o usuário escolha a opção 2 a seguinte operação será realizada:
        print("\nProdutos disponíveis para venda:") # o programa exibe os produtos disponíveis para venda, mostrando o nome do produto e seu preço
        for produto, preco in ld_produtos.items(): # o "for" cria um laço de repetição para percorrer o dicionário de produtos, onde "produto" representa o nome do produto e "preco" representa o preço
            print(f"{produto}: R${preco:.2f}") # o programa exibe o nome do produto e seu preço

    elif li_opcao == "3": # caso o usuário escolha a opção 3 a seguinte operação será realizada:
        print("Encerrando o programa. Até mais!") # o programa exibe uma mensagem
        break # e logo em seguida encerra o programa utilizando o comando break, que encerra o loop do while

    else: # se o usuário digitar uma opção inválida, ou seja, diferente de 1, 2 ou 3
        print("Opção inválida. Por favor, escolha uma opção válida.") # o programa exibe uma mensagem de erro, orientando o usuário a escolher uma opção válida



print("\nRelatório de Vendas") # o programa exibe um relatório de vendas, mostrando os lucros obtidos com cada produto e o lucro total obtido com as vendas
print("Lucros obtidos com as vendas:") # o programa exibe os lucros obtidos com as vendas, mostrando o nome do produto e o lucro correspondente
for produto, lucro in ld_lucro.items(): # o "for" cria um laço de repetição para percorrer o dicionário de lucros
    print(f"{produto}: R${lucro:.2f}") # o programa exibe o nome do produto e o lucro correspondente, 
total_lucro = sum(ld_lucro.values()) # o programa calcula o lucro total obtido com as vendas utilizando a função sum() para somar os valores do dicionário de lucros
